Prime numbers 1 to 100

Question: Write a JavaScript program to display all prime numbers between 1 and 100 in an HTML page.

View source code (HTML + JavaScript)
<div id="out" class="chips"></div>
<p id="count"></p>

<script>
function isPrime(n) {
  if (n < 2) return false;
  for (let i = 2; i * i <= n; i++) {
    if (n % i === 0) return false;
  }
  return true;
}
const out = document.getElementById("out");
let count = 0;
for (let i = 1; i <= 100; i++) {
  if (isPrime(i)) {
    out.innerHTML += '<span class="chip">' + i + "</span>";
    count++;
  }
}
document.getElementById("count").textContent = "Total prime numbers: " + count;
</script>
