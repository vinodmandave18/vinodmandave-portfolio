Client-Side Web Technologies · Lab assignment 21

Replace letter m with a

Question: Write a JavaScript program to find all occurrences of the letter m in a user-entered string, replace them with a, and display the modified string on the page.

View source code (HTML + JavaScript)
<label for="txt">Enter a string</label>
<input id="txt" value="my mom makes marvelous mango milkshake">
<button id="go">Find and replace</button>
<div id="out"></div>

<script>
document.getElementById("go").addEventListener("click", function () {
  const s = document.getElementById("txt").value;
  const out = document.getElementById("out");
  if (s === "") { out.innerHTML = '<div class="msg show err">Please enter a string.</div>'; return; }
  let count = 0, positions = [], result = "";
  for (let i = 0; i < s.length; i++) {
    if (s[i] === "m" || s[i] === "M") {      // find every m (either case)
      count++;
      positions.push(i);
      result += s[i] === "m" ? "a" : "A";    // replace with a
    } else {
      result += s[i];
    }
  }
  const p = document.createElement("div");
  p.className = "msg show ok";
  p.innerHTML = "<strong>Original:</strong> <span id=o></span><br><strong>Occurrences of m:</strong> " + count +
    (count ? " (at index " + positions.join(", ") + ")" : "") + "<br><strong>Modified:</strong> <span id=r></span>";
  out.innerHTML = ""; out.appendChild(p);
  document.getElementById("o").textContent = s;
  document.getElementById("r").textContent = result;
});
</script>
