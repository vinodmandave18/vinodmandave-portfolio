setInterval() and clearInterval()

Question: Write a JavaScript program demonstrating the use of setInterval() and clearInterval().

0

View source code (HTML + JavaScript)
<div class="big" id="counter">0</div>
<p id="clock"></p>
<button id="start">Start</button><button id="stop" class="alt">Stop (clearInterval)</button><button id="reset" class="alt">Reset</button>
<div id="msg" class="msg"></div>

<script>
let timer = null, count = 0;
const $ = id => document.getElementById(id);
function say(t) { $("msg").className = "msg show ok"; $("msg").textContent = t; }
$("start").onclick = function () {
  if (timer !== null) { say("Already running."); return; }
  timer = setInterval(function () {
    count++;
    $("counter").textContent = count;
    $("clock").textContent = "Tick at " + new Date().toLocaleTimeString();
  }, 1000);
  say("setInterval started: runs every 1000 ms.");
};
$("stop").onclick = function () {
  clearInterval(timer);
  timer = null;
  say("clearInterval called: counter stopped at " + count + ".");
};
$("reset").onclick = function () { clearInterval(timer); timer = null; count = 0; $("counter").textContent = 0; $("clock").textContent = ""; say("Reset."); };
</script>
