Online examination form

Question: Design an online examination form using HTML and JavaScript.

View source code (HTML + JavaScript)
<form id="examForm" novalidate>
  <div class="row">
    <div><label for="sname">Student name</label><input id="sname" placeholder="Full name"></div>
    <div><label for="roll">Roll number</label><input id="roll" placeholder="e.g. 101"></div>
  </div>
  <div id="questions"></div>
  <button type="submit">Submit exam</button>
  <button type="reset" class="alt" id="resetBtn">Reset</button>
</form>
<div id="result" class="msg"></div>

<script>
const questions = [
  {q:"Which tag is used to include a JavaScript file?", o:["<link>","<script>","<js>","<code>"], a:1},
  {q:"Which method shows a message box with an OK button?", o:["alert()","prompt()","print()","log()"], a:0},
  {q:"Which symbol starts a single-line comment in JavaScript?", o:["<!-- -->","#","//","**"], a:2},
  {q:"Which keyword declares a block-scoped variable?", o:["var","let","int","dim"], a:1},
  {q:"What does typeof 42 return?", o:["string","number","integer","object"], a:1}
];
const box = document.getElementById("questions");
questions.forEach((item, i) => {
  const wrap = document.createElement("div");
  const h = document.createElement("label");
  h.textContent = (i + 1) + ". " + item.q;
  wrap.appendChild(h);
  item.o.forEach((text, j) => {
    const l = document.createElement("label");
    l.className = "inline";
    const r = document.createElement("input");
    r.type = "radio"; r.name = "q" + i; r.value = j;
    l.appendChild(r);
    l.appendChild(document.createTextNode(text));
    wrap.appendChild(l);
  });
  box.appendChild(wrap);
});
const result = document.getElementById("result");
function show(text, ok) { result.textContent = text; result.className = "msg show " + (ok ? "ok" : "err"); }
document.getElementById("examForm").addEventListener("submit", function (e) {
  e.preventDefault();
  const name = document.getElementById("sname").value.trim();
  const roll = document.getElementById("roll").value.trim();
  if (name === "" || roll === "") { show("Please enter your name and roll number.", false); return; }
  let score = 0;
  for (let i = 0; i < questions.length; i++) {
    const chosen = document.querySelector('input[name="q' + i + '"]:checked');
    if (!chosen) { show("Please answer question " + (i + 1) + ".", false); return; }
    if (Number(chosen.value) === questions[i].a) score++;
  }
  show(name + " (Roll " + roll + "): you scored " + score + " out of " + questions.length + " - " + (score >= 3 ? "Pass" : "Fail"), score >= 3);
});
document.getElementById("resetBtn").addEventListener("click", () => result.className = "msg");
</script>
