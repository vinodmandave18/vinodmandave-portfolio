Hospital registration form

Question: Design a hospital registration form using HTML and JavaScript.

View source code (HTML + JavaScript)
<form id="hForm" novalidate>
  <div class="row">
    <div><label for="pname">Patient name</label><input id="pname"></div>
    <div><label for="age">Age</label><input id="age" type="number" min="0" max="120"></div>
  </div>
  <div class="row">
    <div><label for="phone">Mobile number</label><input id="phone" maxlength="10"></div>
    <div><label for="blood">Blood group</label>
      <select id="blood"><option value="">-- Select --</option><option>A+</option><option>A-</option><option>B+</option><option>B-</option><option>AB+</option><option>AB-</option><option>O+</option><option>O-</option></select></div>
  </div>
  <div class="row">
    <div><label for="dept">Department</label>
      <select id="dept"><option value="">-- Select --</option><option>General Medicine</option><option>Cardiology</option><option>Orthopedics</option><option>Pediatrics</option><option>ENT</option></select></div>
    <div><label for="adate">Appointment date</label><input id="adate" type="date"></div>
  </div>
  <label>Gender</label>
  <label class="inline"><input type="radio" name="g" value="Male">Male</label>
  <label class="inline"><input type="radio" name="g" value="Female">Female</label>
  <label class="inline"><input type="radio" name="g" value="Other">Other</label>
  <label for="symptoms">Symptoms / reason for visit</label><textarea id="symptoms" rows="3"></textarea>
  <button type="submit">Register patient</button><button type="reset" class="alt">Reset</button>
</form>
<div id="msg" class="msg"></div>

<script>
const $ = id => document.getElementById(id);
$("adate").min = new Date().toISOString().split("T")[0];
$("hForm").addEventListener("submit", function (e) {
  e.preventDefault();
  const m = $("msg");
  const g = document.querySelector('input[name="g"]:checked');
  const age = Number($("age").value);
  let error = "";
  if ($("pname").value.trim() === "") error = "Patient name is required.";
  else if (!age || age < 0 || age > 120) error = "Enter a valid age (1-120).";
  else if (!g) error = "Select a gender.";
  else if (!/^[6-9][0-9]{9}$/.test($("phone").value)) error = "Enter a valid 10-digit mobile number.";
  else if ($("blood").value === "") error = "Select a blood group.";
  else if ($("dept").value === "") error = "Select a department.";
  else if ($("adate").value === "") error = "Select an appointment date.";
  if (error) { m.className = "msg show err"; m.textContent = error; return; }
  const id = "HSP" + Math.floor(1000 + Math.random() * 9000);
  m.className = "msg show ok";
  m.textContent = "Registered! Patient ID: " + id + " | " + $("pname").value.trim() + ", " + age + " yrs | " + $("dept").value + " on " + $("adate").value;
});
</script>
