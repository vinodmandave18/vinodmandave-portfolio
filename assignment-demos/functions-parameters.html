Functions with parameter passing

Question: Write a JavaScript program demonstrating functions with parameter passing.

View source code (HTML + JavaScript)
<div class="row">
  <div><label for="n1">Number 1</label><input id="n1" type="number" value="6"></div>
  <div><label for="n2">Number 2</label><input id="n2" type="number" value="3"></div>
</div>
<label for="nm">Name</label><input id="nm" value="Yash">
<button id="b1">add(a, b)</button>
<button id="b2" class="alt">power(base, exponent)</button>
<button id="b3" class="alt">greet(name, times)</button>
<div id="out" class="msg"></div>

<script>
// Function with two parameters
function add(a, b) { return a + b; }
// Function with two parameters, using Math.pow
function power(base, exponent) { return Math.pow(base, exponent); }
// Function with a default parameter value
function greet(name, times = 1) {
  let s = "";
  for (let i = 0; i < times; i++) s += "Hello, " + name + "! ";
  return s;
}
const $ = id => document.getElementById(id);
function show(t) { $("out").textContent = t; $("out").className = "msg show ok"; }
$("b1").onclick = () => show("add(" + $("n1").value + ", " + $("n2").value + ") = " + add(Number($("n1").value), Number($("n2").value)));
$("b2").onclick = () => show("power(" + $("n1").value + ", " + $("n2").value + ") = " + power(Number($("n1").value), Number($("n2").value)));
$("b3").onclick = () => show(greet($("nm").value, Number($("n2").value) || 1));
</script>
