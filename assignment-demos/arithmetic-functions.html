Client-Side Web Technologies · Lab assignment 19

Arithmetic operations using functions

Question: Create a web page with two text boxes and JavaScript functions to perform addition, subtraction, multiplication, and division.

View source code (HTML + JavaScript)
<div class="row">
  <div><label for="a">First number</label><input id="a" type="number"></div>
  <div><label for="b">Second number</label><input id="b" type="number"></div>
</div>
<button onclick="calc('add')">Add</button><button onclick="calc('sub')" class="alt">Subtract</button>
<button onclick="calc('mul')" class="alt">Multiply</button><button onclick="calc('div')" class="alt">Divide</button>
<div id="out" class="msg"></div>

<script>
function addition(a, b) { return a + b; }
function subtraction(a, b) { return a - b; }
function multiplication(a, b) { return a * b; }
function division(a, b) { return a / b; }
function calc(op) {
  const sa = document.getElementById("a").value, sb = document.getElementById("b").value;
  const out = document.getElementById("out");
  if (sa === "" || sb === "") { out.className = "msg show err"; out.textContent = "Please enter both numbers."; return; }
  const a = Number(sa), b = Number(sb);
  let r, sym;
  if (op === "add") { r = addition(a, b); sym = "+"; }
  else if (op === "sub") { r = subtraction(a, b); sym = "-"; }
  else if (op === "mul") { r = multiplication(a, b); sym = "x"; }
  else {
    if (b === 0) { out.className = "msg show err"; out.textContent = "Cannot divide by zero."; return; }
    r = division(a, b); sym = "/";
  }
  out.className = "msg show ok";
  out.textContent = a + " " + sym + " " + b + " = " + r;
}
</script>
