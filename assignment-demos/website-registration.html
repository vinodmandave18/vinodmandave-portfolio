<div id="msg" class="msg"></div>

<script>
const $ = id => document.getElementById(id);
function setErr(n, text) { $("e" + n).textContent = text; return text !== ""; }
$("regForm").addEventListener("submit", function (e) {
  e.preventDefault();
  let bad = false;
  bad = setErr(1, $("fullname").value.trim() === "" ? "Name is required" : "") || bad;
  bad = setErr(2, /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test($("email").value) ? "" : "Enter a valid email") || bad;
  bad = setErr(3, $("pass").value.length < 6 ? "Password must be at least 6 characters" : "") || bad;
  bad = setErr(4, $("cpass").value !== $("pass").value || $("cpass").value === "" ? "Passwords do not match" : "") || bad;
  bad = setErr(5, $("dob").value === "" ? "Select your date of birth" : "") || bad;
  bad = setErr(6, $("country").value === "" ? "Select a country" : "") || bad;
  bad = setErr(7, document.querySelector('input[name="gender"]:checked') ? "" : "Select a gender") || bad;
  bad = setErr(8, $("terms").checked ? "" : "You must accept the terms") || bad;
  const m = $("msg");
  if (bad) { m.className = "msg show err"; m.textContent = "Please correct the errors above."; }
  else { m.className = "msg show ok"; m.textContent = "Registration successful. Welcome, " + $("fullname").value.trim() + "!"; }
});
</script>
