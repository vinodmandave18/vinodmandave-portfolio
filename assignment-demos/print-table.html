Print multiplication table

Question: Create an HTML page with a text box, a Print Table button, and a text area. On clicking the button, display the multiplication table of the entered number. If the input is not a valid number, change the background colour to red and display an alert message.

View source code (HTML + JavaScript)
<label for="num">Enter a number</label>
<input id="num">
<button id="print">Print Table</button>
<label for="area">Table</label>
<textarea id="area" rows="11" readonly></textarea>

<script>
const input = document.getElementById("num");
document.getElementById("print").addEventListener("click", function () {
  const v = input.value.trim();
  const area = document.getElementById("area");
  if (v === "" || isNaN(v)) {
    input.style.background = "red";       // invalid: red background
    input.style.color = "white";
    area.value = "";
    alert("Invalid input! Please enter a valid number.");
    return;
  }
  input.style.background = "";
  input.style.color = "";
  const n = Number(v);
  let text = "";
  for (let i = 1; i <= 10; i++) {
    text += n + " x " + i + " = " + (n * i) + "\n";
  }
  area.value = text;
});
</script>
