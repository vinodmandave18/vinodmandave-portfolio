Student registration validation

Question: Create a student registration form and validate required fields, email address, and mobile number using JavaScript.

View source code (HTML + JavaScript)
<form id="sForm" novalidate>
  <div class="row">
    <div><label for="fn">First name</label><input id="fn"><div class="err-text" id="e1"></div></div>
    <div><label for="ln">Last name</label><input id="ln"><div class="err-text" id="e2"></div></div>
  </div>
  <label for="em">Email address</label><input id="em"><div class="err-text" id="e3"></div>
  <label for="mb">Mobile number</label><input id="mb" maxlength="10" placeholder="10 digits"><div class="err-text" id="e4"></div>
  <div class="row">
    <div><label for="course">Course</label>
      <select id="course"><option value="">-- Select --</option><option>BBA</option><option>BCA</option><option>MBA</option><option>MCA</option></select>
      <div class="err-text" id="e5"></div></div>
    <div><label for="year">Year</label>
      <select id="year"><option value="">-- Select --</option><option>First</option><option>Second</option><option>Third</option></select>
      <div class="err-text" id="e6"></div></div>
  </div>
  <label for="addr">Address</label><textarea id="addr" rows="2"></textarea><div class="err-text" id="e7"></div>
  <button type="submit">Register</button><button type="reset" class="alt">Reset</button>
</form>
<div id="msg" class="msg"></div>

<script>
const $ = id => document.getElementById(id);
function err(n, t) { $("e" + n).textContent = t; return t !== ""; }
$("sForm").addEventListener("submit", function (e) {
  e.preventDefault();
  let bad = false;
  bad = err(1, $("fn").value.trim() === "" ? "First name is required" : "") || bad;
  bad = err(2, $("ln").value.trim() === "" ? "Last name is required" : "") || bad;
  const email = $("em").value.trim();
  bad = err(3, email === "" ? "Email is required" : (/^[^\s@]+@[^\s@]+\.[a-zA-Z]{2,}$/.test(email) ? "" : "Enter a valid email, e.g. name@example.com")) || bad;
  const mob = $("mb").value.trim();
  bad = err(4, mob === "" ? "Mobile number is required" : (/^[6-9][0-9]{9}$/.test(mob) ? "" : "Enter a valid 10-digit mobile number starting with 6-9")) || bad;
  bad = err(5, $("course").value === "" ? "Select a course" : "") || bad;
  bad = err(6, $("year").value === "" ? "Select a year" : "") || bad;
  bad = err(7, $("addr").value.trim() === "" ? "Address is required" : "") || bad;
  const m = $("msg");
  if (bad) { m.className = "msg show err"; m.textContent = "Please fix the highlighted fields."; }
  else { m.className = "msg show ok"; m.textContent = "Student " + $("fn").value.trim() + " " + $("ln").value.trim() + " registered successfully for " + $("course").value + "."; }
});
</script>
