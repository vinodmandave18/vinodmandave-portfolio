Login page validation

Question: Create a login page with username and password fields. Check whether the input fields are blank and display appropriate messages.

View source code (HTML + JavaScript)
<form id="loginForm" novalidate style="max-width:380px">
  <label for="user">Username</label><input id="user" autocomplete="off">
  <label for="pwd">Password</label><input id="pwd" type="password">
  <button type="submit">Login</button>
</form>
<div id="msg" class="msg"></div>

<script>
document.getElementById("loginForm").addEventListener("submit", function (e) {
  e.preventDefault();
  const u = document.getElementById("user").value.trim();
  const p = document.getElementById("pwd").value;
  const m = document.getElementById("msg");
  m.className = "msg show err";
  if (u === "" && p === "") m.textContent = "Username and password are blank. Please fill both fields.";
  else if (u === "") m.textContent = "Username cannot be blank.";
  else if (p === "") m.textContent = "Password cannot be blank.";
  else { m.className = "msg show ok"; m.textContent = "Login successful. Welcome, " + u + "!"; }
});
</script>
