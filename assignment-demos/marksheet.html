Student marksheet by seat number

Question: Display a student's marksheet based on a specified seat number.

View source code (HTML + JavaScript)
<label for="seat">Enter seat number</label>
<input id="seat" placeholder="Try 1001, 1002, 1003 or 1004">
<button id="go">Show marksheet</button>
<div id="out"></div>

<script>
const students = {
  1001: {name:"Aarav Patil", marks:{"Client-Side Web Technologies":78,"Java Programming":82,"DBMS":69,"Python":74,"Mathematics":65}},
  1002: {name:"Sneha Kulkarni", marks:{"Client-Side Web Technologies":91,"Java Programming":88,"DBMS":85,"Python":90,"Mathematics":79}},
  1003: {name:"Rohan Deshmukh", marks:{"Client-Side Web Technologies":55,"Java Programming":48,"DBMS":62,"Python":51,"Mathematics":39}},
  1004: {name:"Priya Shah", marks:{"Client-Side Web Technologies":67,"Java Programming":72,"DBMS":70,"Python":64,"Mathematics":58}}
};
document.getElementById("go").addEventListener("click", function () {
  const out = document.getElementById("out");
  const seat = document.getElementById("seat").value.trim();
  const s = students[seat];
  if (!s) { out.innerHTML = '<div class="msg show err">No student found for seat number "' + seat.replace(/</g, "") + '".</div>'; return; }
  let rows = "", total = 0, pass = true;
  for (const subject in s.marks) {
    const m = s.marks[subject];
    total += m;
    if (m < 40) pass = false;
    rows += "<tr><td>" + subject + "</td><td>100</td><td>" + m + "</td><td>" + (m >= 40 ? "Pass" : "Fail") + "</td></tr>";
  }
  const max = Object.keys(s.marks).length * 100;
  const pct = (total / max * 100).toFixed(2);
  const grade = !pass ? "F" : pct >= 75 ? "A (Distinction)" : pct >= 60 ? "B (First Class)" : pct >= 50 ? "C (Second Class)" : "D (Pass)";
  out.innerHTML = "<h3>" + s.name + " &mdash; Seat No. " + seat + "</h3><table><tr><th>Subject</th><th>Max</th><th>Obtained</th><th>Status</th></tr>" + rows +
    "<tr><th>Total</th><th>" + max + "</th><th>" + total + "</th><th></th></tr></table><p><strong>Percentage:</strong> " + pct + "% &nbsp; <strong>Result:</strong> " + (pass ? "PASS" : "FAIL") + " &nbsp; <strong>Grade:</strong> " + grade + "</p>";
});
</script>
